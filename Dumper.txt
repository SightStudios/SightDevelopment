local dumpedScripts = {}
local scriptCounter = 0
local dumpFolder = "Prot_Dumper"

if makefolder then
    makefolder(dumpFolder)
end

local oldLoadstring = loadstring
getgenv().loadstring = function(src, chunkname)
    if type(src) == "string" and #src > 200 then
        scriptCounter = scriptCounter + 1
        local timestamp = os.time()
        local fileName = dumpFolder .. "/dump_" .. tostring(scriptCounter) .. "_" .. tostring(timestamp) .. ".txt"
        table.insert(dumpedScripts, {id = scriptCounter, code = src, time = timestamp})
        
        if writefile then
            writefile(fileName, src)
        end
        
        warn("Captured script #" .. tostring(scriptCounter) .. " saved as " .. fileName)
    end
    return oldLoadstring(src, chunkname)
end

local selectedScriptId = nil

local function updateScriptList()
    for _, child in pairs(listBox:GetChildren()) do
        if child:IsA("TextButton") then
            child:Destroy()
        end
    end
    
    title.Text = "Prot Dumper - Captured Scripts: " .. tostring(#dumpedScripts)
    
    for _, script in ipairs(dumpedScripts) do
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(1, -10, 0, 40)
        btn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
        btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        btn.Font = Enum.Font.SourceSans
        btn.TextSize = 12
        btn.Text = "Script #" .. tostring(script.id) .. " - " .. os.date("%H:%M:%S", script.time)
        btn.Parent = listBox
        btn.MouseButton1Click:Connect(function()
            selectedScriptId = script.id
            outputBox.Text = script.code
            for _, otherBtn in pairs(listBox:GetChildren()) do
                if otherBtn:IsA("TextButton") then
                    otherBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
                end
            end
            btn.BackgroundColor3 = Color3.fromRGB(80, 80, 80)
        end)
    end
    
    listBox.CanvasSize = UDim2.new(0, 0, 0, (#dumpedScripts * 45) + 10)
end

copyButton.MouseButton1Click:Connect(function()
    if selectedScriptId then
        for _, script in ipairs(dumpedScripts) do
            if script.id == selectedScriptId then
                if setclipboard then
                    setclipboard(script.code)
                    outputBox.Text = "Copied to clipboard!"
                else
                    outputBox.Text = "Executor does not support clipboard."
                end
                break
            end
        end
    else
        outputBox.Text = "No script selected."
    end
end)

saveButton.MouseButton1Click:Connect(function()
    if selectedScriptId then
        for _, script in ipairs(dumpedScripts) do
            if script.id == selectedScriptId then
                if writefile then
                    local saveName = dumpFolder .. "/manual_save_" .. tostring(script.id) .. "_" .. tostring(os.time()) .. ".txt"
                    writefile(saveName, script.code)
                    outputBox.Text = "Saved to " .. saveName
                else
                    outputBox.Text = "Executor does not support writefile."
                end
                break
            end
        end
    else
        outputBox.Text = "No script selected."
    end
end)

refreshButton.MouseButton1Click:Connect(function()
    updateScriptList()
    outputBox.Text = "Refreshed!"
    task.wait(0.5)
    if selectedScriptId then
        for _, script in ipairs(dumpedScripts) do
            if script.id == selectedScriptId then
                outputBox.Text = script.code
                break
            end
        end
    end
end)

RunService.RenderStepped:Connect(function()
    if #dumpedScripts > 0 and (#listBox:GetChildren() - 1) ~= #dumpedScripts then
        updateScriptList()
    end
end)

warn("Prot Dumper ready. Execute any script now - every capture will be saved to '" .. dumpFolder .. "' folder")
