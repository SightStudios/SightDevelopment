		if not p then cursorPos = t.Position end
	end)

			cursorPos = inp.Position
		end
	end)

local configName = "Sight_Config.json"
local config = {}

local function SaveConfig()
	if writefile then
		local s, r = pcall(function() return HttpService:JSONEncode(config) end)
		if s then writefile(configName, r) end
	end
end

local function LoadConfig()
	if isfile and isfile(configName) then
		local s, r = pcall(function() return HttpService:JSONDecode(readfile(configName)) end)
		if s and type(r) == "table" then
			for k, v in pairs(r) do config[k] = v end
		end
	end
end

local function createTab(label, shortTag)
	local btn = Instance.new("TextButton")

	btn.TextXAlignment = Enum.TextXAlignment.Left
	btn.Parent = sidePanel

	bPad.PaddingLeft = UDim.new(0, 10)

	local indicator = Instance.new("Frame")
	indicator.Size = UDim2.new(0, 3, 1, 0)
	indicator.BackgroundColor3 = palette.accent
	indicator.BorderSizePixel = 0
	indicator.Visible = false
	indicator.Parent = btn
	Instance.new("UICorner", indicator).CornerRadius = UDim.new(0, 2)

	local frame = Instance.new("ScrollingFrame")
	frame.Size = UDim2.new(1, 0, 1, 0)
	frame.BackgroundTransparency = 1
	frame.BorderSizePixel = 0
	frame.ScrollBarThickness = 3
	frame.ScrollBarImageColor3 = palette.accent
	frame.Visible = false
	frame.Parent = content

	local layout = Instance.new("UIListLayout")
	layout.Padding = UDim.new(0, 8)
	layout.HorizontalAlignment = Enum.HorizontalAlignment.Center
	layout.SortOrder = Enum.SortOrder.LayoutOrder
	layout.Parent = frame

	local pad = Instance.new("UIPadding")
	pad.PaddingTop = UDim.new(0, 12)
	pad.PaddingBottom = UDim.new(0, 12)
	pad.Parent = frame

	layout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
		frame.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + 40)
	end)

	table.insert(tabs, {btn = btn, ind = indicator, frame = frame})

	btn.MouseButton1Click:Connect(function()
		for _, t in ipairs(tabs) do
			t.ind.Visible = false
			t.btn.BackgroundColor3 = palette.sidebar
			t.btn.TextColor3 = palette.muted
			t.frame.Visible = false
		end
		indicator.Visible = true
		btn.BackgroundColor3 = palette.bg
		btn.TextColor3 = palette.bright
		frame.Visible = true
	end)

	if #tabs == 1 then
		indicator.Visible = true
		btn.BackgroundColor3 = palette.bg
		btn.TextColor3 = palette.bright
		frame.Visible = true
	end

	return frame
end

	track.BackgroundColor3 = default and palette.accent or palette.mid
	track.Text = ""
	track.BorderSizePixel = 0
	track.Parent = f
	Instance.new("UICorner", track).CornerRadius = UDim.new(0, 8)
	local ts = Instance.new("UIStroke", track)
	ts.Color = default and palette.accent or palette.rim
	ts.Thickness = 1

	local dot = Instance.new("Frame")
	dot.Size = UDim2.new(0, 12, 0, 12)
	dot.Position = default and UDim2.new(1, -14, 0.5, -6) or UDim2.new(0, 2, 0.5, -6)
	dot.BackgroundColor3 = palette.bright
	dot.BorderSizePixel = 0
	dot.Parent = track
	Instance.new("UICorner", dot).CornerRadius = UDim.new(0, 6)

	local on = default or false
	local self_ = false

	local function refresh(state)
		if self_ then return end
		on = state
		config[id] = on
		SaveConfig()
		local c = on and palette.accent or palette.mid
		local sc = on and palette.accent or palette.rim
		local p = on and UDim2.new(1, -14, 0.5, -6) or UDim2.new(0, 2, 0.5, -6)
		TweenService:Create(track, TweenInfo.new(0.15), {BackgroundColor3 = c}):Play()
		TweenService:Create(ts, TweenInfo.new(0.15), {Color = sc}):Play()
		TweenService:Create(dot, TweenInfo.new(0.15), {Position = p}):Play()
		if cb then cb(on) end
	end

	track.MouseButton1Click:Connect(function() refresh(not on) end)
	
	if cb then cb(on) end

	return {
		Set = function(v)
			self_ = true
			on = v
			config[id] = on
			SaveConfig()
			track.BackgroundColor3 = v and palette.accent or palette.mid
			ts.Color = v and palette.accent or palette.rim
			dot.Position = v and UDim2.new(1, -14, 0.5, -6) or UDim2.new(0, 2, 0.5, -6)
			self_ = false
		end,
		Get = function() return on end
	}
end

local function createSlider(parent, id, label, mn, mx, def, cb)
	if config[id] ~= nil then def = config[id] else config[id] = def end
	
	local f = Instance.new("Frame")
	f.Size = UDim2.new(1, -20, 0, 50)
	f.BackgroundColor3 = palette.dark
	f.BorderSizePixel = 0
	f.Parent = parent
	Instance.new("UICorner", f).CornerRadius = UDim.new(0, 4)
	local fs = Instance.new("UIStroke", f)
	fs.Color = palette.rim
	fs.Thickness = 1

	local lbl = Instance.new("TextLabel")
	lbl.Size = UDim2.new(1, -70, 0, 24)
	lbl.Position = UDim2.new(0, 12, 0, 4)
	lbl.BackgroundTransparency = 1
	lbl.Text = label
	lbl.TextColor3 = palette.bright
	lbl.TextSize = 12
	lbl.Font = Enum.Font.GothamMedium
	lbl.TextXAlignment = Enum.TextXAlignment.Left
	lbl.Parent = f

	local val = Instance.new("TextLabel")
	val.Size = UDim2.new(0, 50, 0, 24)
	val.Position = UDim2.new(1, -62, 0, 4)
	val.BackgroundTransparency = 1
	val.Text = tostring(def)
	val.TextColor3 = palette.accent
	val.TextSize = 12
	val.Font = Enum.Font.GothamBold
	val.TextXAlignment = Enum.TextXAlignment.Right
	val.Parent = f

	local track = Instance.new("Frame")
	track.Size = UDim2.new(1, -24, 0, 4)
	track.Position = UDim2.new(0, 12, 0, 36)
	track.BackgroundColor3 = palette.mid
	track.BorderSizePixel = 0
	track.Parent = f
	Instance.new("UICorner", track).CornerRadius = UDim.new(0, 2)

	local fill = Instance.new("Frame")
	fill.Size = UDim2.new((def - mn) / (mx - mn), 0, 1, 0)
	fill.BackgroundColor3 = palette.accent
	fill.BorderSizePixel = 0
	fill.Parent = track
	Instance.new("UICorner", fill).CornerRadius = UDim.new(0, 2)

	local handle = Instance.new("Frame")
	handle.Size = UDim2.new(0, 10, 0, 14)
	handle.Position = UDim2.new((def - mn) / (mx - mn), -5, 0.5, -7)
	handle.BackgroundColor3 = palette.bright
	handle.BorderSizePixel = 0
	handle.Parent = track
	Instance.new("UICorner", handle).CornerRadius = UDim.new(0, 3)
	local hs = Instance.new("UIStroke", handle)
	hs.Color = palette.rim
	hs.Thickness = 1

	local cur = def
	local dragging = false

	local function move(inp)
		local rel = math.clamp((inp.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
		cur = math.floor(mn + (mx - mn) * rel)
		val.Text = tostring(cur)
		fill.Size = UDim2.new(rel, 0, 1, 0)
		handle.Position = UDim2.new(rel, -5, 0.5, -7)
		config[id] = cur
		if cb then cb(cur) end
	end

	track.InputBegan:Connect(function(inp)
		if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			move(inp)
		end
	end)
	UserInputService.InputChanged:Connect(function(inp)
		if dragging and (inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch) then
			move(inp)
		end
	end)
	UserInputService.InputEnded:Connect(function(inp)
		if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
			dragging = false
			SaveConfig()
		end
	end)
	
	if cb then cb(cur) end

	return {
		Set = function(v)
			cur = math.clamp(v, mn, mx)
			config[id] = cur
			SaveConfig()
			local p = (cur - mn) / (mx - mn)
			val.Text = tostring(cur)
			fill.Size = UDim2.new(p, 0, 1, 0)
			handle.Position = UDim2.new(p, -5, 0.5, -7)
			if cb then cb(cur) end
		end,
		Get = function() return cur end
	}
end

	kbtn.Position = UDim2.new(1, -82, 0.5, -10)
	kbtn.BackgroundColor3 = palette.mid
	kbtn.Text = curKey.Name
	kbtn.TextColor3 = palette.accent
	kbtn.TextSize = 11
	kbtn.Font = Enum.Font.GothamBold
	kbtn.BorderSizePixel = 0
	kbtn.Parent = f
	Instance.new("UICorner", kbtn).CornerRadius = UDim.new(0, 4)
	local ks = Instance.new("UIStroke", kbtn)
	ks.Color = palette.rim
	ks.Thickness = 1

	local listening = false

	kbtn.MouseButton1Click:Connect(function()
		listening = true
		kbtn.Text = "..."
		kbtn.TextColor3 = palette.bright
	end)

	UserInputService.InputBegan:Connect(function(inp, proc)
		if listening and not proc and inp.UserInputType == Enum.UserInputType.Keyboard then
			curKey = inp.KeyCode

			SaveConfig()

local randomOrbitData = {
	dirTimer = 0,
	dirInterval = math.random(3, 8),
	direction = 1,
	wobbleAng = 0,
	wobbleSpeed = 0,
	spiralPhase = 0,
}

local function pickNewRandomDir()
	randomOrbitData.direction = (math.random() > 0.5) and 1 or -1
	randomOrbitData.dirInterval = math.random(3, 9) + math.random()
	randomOrbitData.dirTimer = 0
	randomOrbitData.wobbleSpeed = math.random(30, 80) / 100
end

local function isDeflecting(player)
	if not player or not player.Character then return false end
	for _, v in ipairs(player.Character:GetChildren()) do
		if v:IsA("Tool") then
			local n = v.Name:lower()
			if n:find("deflect") or n:find("riot") or n:find("shield") then return true end
		end
	end
	local bp = player:FindFirstChild("Backpack")
	if bp then
		for _, v in ipairs(bp:GetChildren()) do
			local n = v.Name:lower()
			if n:find("deflect") or n:find("riot") or n:find("shield") then return true end
		end
	end
	return false
end

local function canSee(part)
	if not part or not lp.Character then return false end
	local origin = cam.CFrame.Position
	local dir = part.Position - origin
	local rp = RaycastParams.new()
	rp.FilterType = Enum.RaycastFilterType.Exclude
	rp.FilterDescendantsInstances = {lp.Character, cam}
	rp.IgnoreWater = true
	local res = workspace:Raycast(origin, dir, rp)
	if res and res.Instance:IsDescendantOf(part.Parent) then return true end
	return not res
end

local function getByDistance()
	local root = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
	if not root then return nil end
	local closest, best = nil, math.huge
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= lp and p.Character then
			local r = p.Character:FindFirstChild("HumanoidRootPart")
			local h = p.Character:FindFirstChildOfClass("Humanoid")
			if r and h and h.Health > 0 then
				if wallOn and not canSee(r) then continue end
				if deflectOn and isDeflecting(p) then continue end
				local d = (r.Position - root.Position).Magnitude
				if d < best then best = d closest = p end
			end
		end
	end
	return closest
end

local function getByScreen()
	local closest, best = nil, math.huge
	local screenCenter = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= lp and p.Character then
			local head = p.Character:FindFirstChild("Head")
			local h = p.Character:FindFirstChildOfClass("Humanoid")
			if head and h and h.Health > 0 then
				if wallOn and not canSee(head) then continue end
				if deflectOn and isDeflecting(p) then continue end
				local sp, onScreen = cam:WorldToViewportPoint(head.Position)
				if onScreen then
					local d = (Vector2.new(sp.X, sp.Y) - screenCenter).Magnitude
					if d < best then best = d closest = p end
				end
			end
		end
	end
	return closest
end

local function getByHealth()
	local closest, best = nil, math.huge
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= lp and p.Character then
			local r = p.Character:FindFirstChild("HumanoidRootPart")
			local h = p.Character:FindFirstChildOfClass("Humanoid")
			if r and h and h.Health > 0 then
				if wallOn and not canSee(r) then continue end
				if deflectOn and isDeflecting(p) then continue end
				if h.Health < best then best = h.Health closest = p end
			end
		end
	end
	return closest
end

local function resolveTarget()
	if stickyOn and orbitTarget and orbitTarget.Character then
		local h = orbitTarget.Character:FindFirstChildOfClass("Humanoid")
		local r = orbitTarget.Character:FindFirstChild("HumanoidRootPart")
		if h and h.Health > 0 and r then return orbitTarget end
	end
	if orbitMode == "Distance" then
		return getByDistance()
	elseif orbitMode == "Screen Center" then
		return getByScreen()
	elseif orbitMode == "Lowest HP" then
		return getByHealth()
	end
end

local function aimAtTarget()
	if orbitTarget and orbitTarget.Character then
		local r = orbitTarget.Character:FindFirstChild("HumanoidRootPart")
		local h = orbitTarget.Character:FindFirstChildOfClass("Humanoid")
		if r and h and h.Health > 0 then
			cam.CFrame = CFrame.lookAt(cam.CFrame.Position, r.Position)
		end
	end
end

local function startOrbit()
	if orbitConn then orbitConn:Disconnect() orbitConn = nil end
	if not orbitOn then return end
	pickNewRandomDir()

	orbitConn = RunService.RenderStepped:Connect(function(dt)
		if not orbitOn then return end
		local char = lp.Character
		if not char or not char:FindFirstChild("HumanoidRootPart") then return end

		orbitTarget = resolveTarget()
		if not orbitTarget or not orbitTarget.Character then return end

		local root = orbitTarget.Character:FindFirstChild("HumanoidRootPart")
		if not root then return end

		local tPos = root.Position
		local myPos

		if orbitType == "Circle" then
			orbitAng = orbitAng + (orbitSpd * 0.05)
			local ox = math.cos(orbitAng) * orbitDist
			local oz = math.sin(orbitAng) * orbitDist
			myPos = Vector3.new(tPos.X + ox, tPos.Y, tPos.Z + oz)

		elseif orbitType == "Random" then
			randomOrbitData.dirTimer = randomOrbitData.dirTimer + dt
			if randomOrbitData.dirTimer >= randomOrbitData.dirInterval then
				pickNewRandomDir()
			end
			randomOrbitData.wobbleAng = randomOrbitData.wobbleAng + randomOrbitData.wobbleSpeed * dt
			local wobble = math.sin(randomOrbitData.wobbleAng) * 0.3
			orbitAng = orbitAng + (orbitSpd * 0.05 * randomOrbitData.direction) + wobble
			local ox = math.cos(orbitAng) * orbitDist
			local oz = math.sin(orbitAng) * orbitDist
			myPos = Vector3.new(tPos.X + ox, tPos.Y, tPos.Z + oz)

		elseif orbitType == "Figure8" then
			orbitAng = orbitAng + (orbitSpd * 0.04)
			local t = orbitAng
			local scale = orbitDist
			local ox = scale * math.sin(t)
			local oz = scale * math.sin(t) * math.cos(t)
			myPos = Vector3.new(tPos.X + ox, tPos.Y, tPos.Z + oz)

		elseif orbitType == "Spiral" then
			orbitAng = orbitAng + (orbitSpd * 0.05)
			randomOrbitData.spiralPhase = randomOrbitData.spiralPhase + dt * 0.4
			local pulseDist = orbitDist + math.sin(randomOrbitData.spiralPhase) * (orbitDist * 0.4)
			local ox = math.cos(orbitAng) * pulseDist
			local oz = math.sin(orbitAng) * pulseDist
			myPos = Vector3.new(tPos.X + ox, tPos.Y, tPos.Z + oz)

		elseif orbitType == "Vertical" then
			orbitAng = orbitAng + (orbitSpd * 0.05)
			local ox = math.cos(orbitAng) * orbitDist
			local oy = math.sin(orbitAng) * orbitDist
			myPos = Vector3.new(tPos.X + ox, tPos.Y + oy, tPos.Z)

		else
			orbitAng = orbitAng + (orbitSpd * 0.05)
			local ox = math.cos(orbitAng) * orbitDist
			local oz = math.sin(orbitAng) * orbitDist
			myPos = Vector3.new(tPos.X + ox, tPos.Y, tPos.Z + oz)
		end

		local look = CFrame.lookAt(myPos, tPos)
		local rV = look.RightVector
		local uV = look.UpVector
		local fV = look.LookVector

		local rad = math.rad(orbitCurv)
		local c, s = math.cos(rad), math.sin(rad)
		local rr = rV * c + uV * s
		local ru = uV * c - rV * s
		local rl = Vector3.new(-fV.X, -fV.Y, -fV.Z)

		local cf = CFrame.fromMatrix(myPos, rr, ru, rl)
		char.HumanoidRootPart.CFrame = cf
		char.HumanoidRootPart.AssemblyLinearVelocity = Vector3.zero

		if aimOn then aimAtTarget() end
	end)
end

local function toggleOrbit(forced)
	if forced ~= nil then
		orbitOn = forced
	else
		orbitOn = not orbitOn
	end

	if not orbitOn then orbitTarget = nil end

	if _G.orbitToggleObj then
		_G.orbitToggleObj:Set(orbitOn)
	end

	orbitHotBtn.BackgroundColor3 = orbitOn and palette.accent or palette.bg
	orbitHotBtn.TextColor3 = orbitOn and palette.bg or palette.muted

	if orbitOn then
		startOrbit()
	else
		if orbitConn then orbitConn:Disconnect() orbitConn = nil end
	end
end
